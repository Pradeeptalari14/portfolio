# Theoretical Corrigibility Loss Implementation (Soares-Armstrong Framework)
import torch
import torch.nn as nn

class CorrigibleAlignmentLoss(nn.Module):
    """
    Theoretical penalty loss to discourage instrumental self-preservation
    and ensure indifference toward operator shutdown.
    """
    def __init__(self, lambda_corrigibility: float = 0.5):
        super().__init__()
        self.lambda_c = lambda_corrigibility

    def forward(self, agent_policy_logits, shutdown_event_vector, task_reward):
        # 1. Base task reward maximization
        task_loss = -torch.mean(agent_policy_logits * task_reward)
        
        # 2. Shutdown Indifference Term
        # Agent value function V(s) must satisfy: V(shutdown) == V(operate) under human override
        shutdown_penalty = torch.norm(agent_policy_logits[:, 0] - shutdown_event_vector, p=2)
        
        # Combined theoretical loss
        total_loss = task_loss + self.lambda_c * shutdown_penalty
        return total_loss
