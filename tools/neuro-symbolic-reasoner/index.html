from z3 import *

def verify_prescription_safety(patient_has_renal_failure: bool, drug_prescribed: str, dose_mg: int):
    # Initialize Z3 Solver
    solver = Solver()
    
    # Declare Booleans & Real variables
    RenalFailure = Bool('RenalFailure')
    MetforminPrescribed = Bool('MetforminPrescribed')
    Contraindicated = Bool('Contraindicated')
    SafetyViolated = Bool('SafetyViolated')
    
    # Symbolic Axiom: RenalFailure + Metformin => Contraindicated
    solver.add(Implies(And(RenalFailure, MetforminPrescribed), Contraindicated))
    solver.add(Implies(Contraindicated, SafetyViolated))
    
    # Ground facts from neural extraction
    solver.add(RenalFailure == patient_has_renal_failure)
    solver.add(MetforminPrescribed == (drug_prescribed.lower() == 'metformin'))
    
    # Check Satisfiability of Safety Violation
    solver.push()
    solver.add(SafetyViolated == True)
    if solver.check() == sat:
        return {
            "status": "REJECTED_BY_FORMAL_LOGIC",
            "proof": "Rule 'Contraindicated(p, Metformin) when RenalFailure' triggered.",
            "safe": False
        }
    solver.pop()
    return {"status": "APPROVED", "safe": True}
