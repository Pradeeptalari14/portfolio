# Anthropic-Style Top-k Sparse Autoencoder (SAE) in PyTorch
import torch
import torch.nn as nn

class TopKSAE(nn.Module):
    def __init__(self, d_model: int = 4096, d_sae: int = 65536, k: int = 32):
        super().__init__()
        self.k = k
        self.encoder = nn.Linear(d_model, d_sae, bias=False)
        self.decoder = nn.Linear(d_sae, d_model, bias=False)
        self.bias = nn.Parameter(torch.zeros(d_model))

    def forward(self, x):
        # Center activations
        x_centered = x - self.bias
        
        # 1. Project to high-dimensional latent space
        pre_acts = nn.functional.relu(self.encoder(x_centered))
        
        # 2. Top-k Sparsification: keep only top k largest activations
        val, idx = torch.topk(pre_acts, self.k, dim=-1)
        latents = torch.zeros_like(pre_acts).scatter_(-1, idx, val)
        
        # 3. Reconstruct dense representation
        x_reconstructed = self.decoder(latents) + self.bias
        return x_reconstructed, latents
